cmake_minimum_required(VERSION 3.31)
project(nws_consumer LANGUAGES CXX)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -DNWS_PARENT_FLAGS")
if(NWS_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(nws_cpp SOURCE_DIR "${NWS_SOURCE_DIR}")
    FetchContent_MakeAvailable(nws_cpp)
    if(NWS_BUILD_TESTS OR NWS_BUILD_EXAMPLES OR NWS_ENABLE_LTO)
        message(FATAL_ERROR "SDK subproject defaults must be off")
    endif()
    get_directory_property(sdk_release_flags DIRECTORY "${NWS_SOURCE_DIR}"
        DEFINITION CMAKE_CXX_FLAGS_RELEASE)
    if(NOT sdk_release_flags STREQUAL "-O2 -DNDEBUG -DNWS_PARENT_FLAGS" OR
       NOT CMAKE_CXX_FLAGS_RELEASE STREQUAL "-O2 -DNDEBUG -DNWS_PARENT_FLAGS")
        message(FATAL_ERROR "SDK overwrote the parent's release flags")
    endif()
else()
    find_package(nws ${NWS_EXPECTED_VERSION} EXACT CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE nws::nws)
